本项目使用静态HTML/JS/CSS前端（位于 `frontend/static/`），主要技术栈：

- Vanilla JavaScript (原生JS)

- ✅ 用户登录/登出

- ✅ 音频修复（降噪、去混响、超分辨率）
- ✅ 参考音频管理

"""

import pytest
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from webdriver_manager.chrome import ChromeDriverManager


@pytest.fixture(scope="module")
def driver():
    """创建浏览器驱动"""
    driver = webdriver.Chrome(ChromeDriverManager().install())
    driver.implicitly_wait(10)
    yield driver
    driver.quit()


class TestLoginFlow:
    """登录流程测试"""

    def test_login_success(self, driver):
        """测试成功登录"""
        driver.get("http://localhost:8000")
        
        # 等待登录表单
        wait = WebDriverWait(driver, 10)
        login_form = wait.until(EC.presence_of_element_located((By.ID, "login-form")))
        
        # 输入登录信息
        username = driver.find_element(By.ID, "username-input")
        password = driver.find_element(By.ID, "password-input")
        login_btn = driver.find_element(By.ID, "login-btn")
        
        username.send_keys("admin")
        password.send_keys("tp123456")
        login_btn.click()
        
        # 等待跳转
        app_section = wait.until(EC.visibility_of_element_located((By.ID, "app-section")))

        user_display = driver.find_element(By.ID, "user-display")

        """测试登出"""
        # 先登录

## 常见问题排查

### Q1: 登录后立即退出
**原因**: Token过期或401错误处理不当  
**排查**: 
```javascript
// 检查localStorage
localStorage.getItem('token')
// 检查Console错误
```

### Q2: 文件上传无进度
**原因**: WebSocket连接失败  
**排查**:
- 检查WebSocket连接状态
- 查看Network面板WS标签
- 检查后端WebSocket路由

### Q3: Toast提示不显示
**原因**: toast-container未创建  
**排查**:
```javascript
// 检查DOM
document.getElementById('toast-container')
```

### Q4: 音频波形不显示
**原因**: WaveSurfer.js初始化失败  
**排查**:
- 检查Console错误
- 验证音频文件格式
- 检查Canvas元素
